function pickWeighted(e){
  let tot=0;for(const x of e)tot+=x[1];
  let r=Math.random()*tot;

}

function finishTrade(id,accept){
  const t=S.trades.find(x=>x.id===id);if(!t)return;
  if(!accept){S.trades=S.trades.filter(x=>x.id!==id);save();renderTrades();toast('Trade proposal closed');return}
  if(t.direction==='incoming'){
    const k=t.offerKey;if(!k||!S.inv[k]){toast('You no longer own the offered card');return}
    S.inv[k]--;if(S.inv[k]<=0)delete S.inv[k];const wantKey=t.wantKey||t.wantRank+':normal';S.inv[wantKey]=(S.inv[wantKey]||0)+1;
  }else{
    if(!S.inv[t.offerKey]){toast('You no longer own the offered card');return}
    S.inv[t.offerKey]--;if(S.inv[t.offerKey]<=0)delete S.inv[t.offerKey];
    const wantKey=t.wantRank+':normal';S.inv[wantKey]=(S.inv[wantKey]||0)+1;
  }
  S.completedTrades.push(t);S.trades=S.trades.filter(x=>x.id!==id);recalc();save();renderCurrent();toast('Trade completed! Cards exchanged.');
}
